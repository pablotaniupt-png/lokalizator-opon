let selectedSpeedIdx=''; let selectedHomologations=[]; function parseQuickTireSpec(raw){ let s=String(raw||'').trim(); if(!s&&!selectedSpeedIdx&&!selectedHomologations.length)return {size:'',autoNote:''}; // 1. Wyciągnij bieżnik z przecinkiem lub kropką (np. 6,3 lub 6.3) let treadStr=''; s=s.replace(/(?:^|\s)(\d{1,2}[,\.]\d)\s*(?:mm)?(?=\s|$)/i, (_,val)=>{ const num=parseFloat(val.replace(',','.')); if(!isNaN(num)&&num>0&&num<25){ const min=Math.max(0,num-0.5).toFixed(1).replace('.',','); const max=(num+0.5).toFixed(1).replace('.',','); const mid=num.toFixed(1).replace('.',','); treadStr='Bieżnik: '+mid+' mm ('+min+'–'+max+' mm)'; return ' '; } return ' '+val+' '; }); // 2. Sformatuj rozmiar (np. 2753520 -> 275/35 R20) s=s.replace(/^(\d{3})(\d{2})(\d{2})(?=\s|$|[A-Za-z*])/, '$1/$2 R$3'); s=s.replace(/^(\d{3})[\/\-](\d{2})[\/\-]?R?(\d{2})/i, '$1/$2 R$3'); // 3. Wyciągnij 2-cyfrowy lub 4-cyfrowy rok DOT na końcu (np. 20 -> DOT 2020) let dotStr=''; s=s.replace(/(?:^|\s)(20\d{2}|\d{2})\s*$/, (_,yr)=>{ const fullYear=yr.length===2?'20'+yr:yr; dotStr='DOT '+fullYear; return ''; }); // 4. Połącz nośność (np. 102) z wybranym kafelkiem prędkości (np. Y -> 102Y) i homologacjami (*, RSC, MO, MOE, MO1) s=s.replace(/\s+/g,' ').trim(); if(selectedSpeedIdx){ if(/\b\d{2,3}$/.test(s)){ s=s+selectedSpeedIdx; }else if(s){ s=s+' '+selectedSpeedIdx; }else{ s=selectedSpeedIdx; } } if(selectedHomologations.length){ s=(s?s+' ':'')+selectedHomologations.join(' '); } const autoNote=[dotStr,treadStr].filter(Boolean).join(' • '); return {size:s.trim(),autoNote}; } function updateTirePreview(){ const parsed=parseQuickTireSpec($('#quickSize')?.value||''); const prev=$('#quickTirePreview'); if(!prev)return; if(!parsed.size&&!parsed.autoNote){prev.textContent='';return;} prev.textContent='Podgląd: '+(parsed.size||'—')+(parsed.autoNote?' | '+parsed.autoNote:''); } function resetQuickTireAll(){ const form=$('#quickTireForm'); if(form)form.reset(); selectedSpeedIdx=''; selectedHomologations=[]; document.querySelectorAll('[data-speed-idx],[data-homologation]').forEach(b=>b.classList.remove('primary')); if($('#quickTirePreview'))$('#quickTirePreview').textContent=''; $('#quickBrand')?.focus(); } $('#resetQuickTireBtn')?.addEventListener('click',resetQuickTireAll); document.querySelectorAll('[data-tire-brand]').forEach(btn=>{ btn.addEventListener('click',()=>{ const brandInp=$('#quickBrand'); const sizeInp=$('#quickSize'); if(brandInp)brandInp.value=btn.dataset.tireBrand; if(sizeInp){sizeInp.focus();} }); }); document.querySelectorAll('[data-speed-idx]').forEach(btn=>{ btn.addEventListener('click',()=>{ const tag=btn.dataset.speedIdx; document.querySelectorAll('[data-speed-idx]').forEach(b=>b.classList.remove('primary')); if(selectedSpeedIdx===tag){ selectedSpeedIdx=''; }else{ selectedSpeedIdx=tag; btn.classList.add('primary'); } updateTirePreview(); }); }); document.querySelectorAll('[data-homologation]').forEach(btn=>{ btn.addEventListener('click',()=>{ const tag=btn.dataset.homologation; const idx=selectedHomologations.indexOf(tag); if(idx>=0){ selectedHomologations.splice(idx,1); btn.classList.remove('primary'); }else{ selectedHomologations.push(tag); btn.classList.add('primary'); } updateTirePreview(); }); }); $('#quickSize')?.addEventListener('input',updateTirePreview); $('#quickTireForm').addEventListener('submit',async function(e){ e.preventDefault(); const parsed=parseQuickTireSpec($('#quickSize').value); const brand=$('#quickBrand').value.trim(); if(!brand&&!parsed.size){showAppToast('Wybierz markę lub wpisz rozmiar opony.','error');return;} const autoSerial='OP-'+Date.now().toString().slice(-6)+Math.floor(10+Math.random()*89); const manualNotes=$('#quickNotes').value.trim(); const fullNotes=[parsed.autoNote,manualNotes].filter(Boolean).join(' • '); // Wyciągnij sam rozmiar bazowy (np. "275/35 R20") do zostawienia na następną oponę const baseSizeMatch=parsed.size.match(/^(\d{3}\/\d{2}\s+R\d{2})/i); const keptBaseSize=baseSizeMatch?baseSizeMatch[0]+' ':''; const tire=normalizeTire({id:uid(),serial:autoSerial,brand,size:parsed.size,status:'Na magazynie',location:'Magazyn',notes:fullNotes,createdAt:new Date().toISOString(),updatedAt:new Date().toISOString()}); try{ setSyncStatus('Zapisywanie opony…'); await addRemote(tire); selectedSpeedIdx=''; selectedHomologations=[]; document.querySelectorAll('[data-speed-idx],[data-homologation]').forEach(b=>b.classList.remove('primary')); $('#quickBrand').value=''; $('#quickSize').value=keptBaseSize; $('#quickNotes').value=''; updateTirePreview(); $('#quickBrand').focus(); await syncFromCloud({allowMigration:false}); showAppToast('Zapisano (zostawiono rozmiar)'); } catch(e){showAppToast('Nie udało się zapisać opony: '+e.message,'error');setSyncStatus('Nie udało się zapisać zmian.');} });
